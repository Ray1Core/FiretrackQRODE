-- ============================================================
-- FIRE TRACK DATABASE SCHEMA (SQL Server – Revised)
-- ============================================================
-- Changes:
--   - TEXT → VARCHAR(MAX), DATETIME → DATETIME2
--   - Added categories lookup table
--   - equipment.category now references category_id
--   - Added indexes on all foreign keys
--   - Added triggers to auto-update updated_at
--   - Increased profile_image_path to NVARCHAR(500)
--   - Explicit ON DELETE NO ACTION for safety
-- ============================================================

USE master;
GO

-- Drop existing database if it exists (CAUTION: removes all data)
IF EXISTS (SELECT name FROM sys.databases WHERE name = N'FiretrackDB')
    DROP DATABASE FiretrackDB;
GO

CREATE DATABASE FiretrackDB;
GO

USE FiretrackDB;
GO

-- ============================================================
-- 1. CATEGORIES lookup table (new)
-- ============================================================
CREATE TABLE categories (
    category_id INT IDENTITY(1,1) PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL UNIQUE,
    description VARCHAR(MAX) NULL,
    created_at DATETIME2 DEFAULT GETDATE()
);
GO

-- ============================================================
-- 2. ROLES
-- ============================================================
CREATE TABLE roles (
    role_id INT IDENTITY(1,1) PRIMARY KEY,
    role_name VARCHAR(50) NOT NULL UNIQUE,
    description VARCHAR(MAX) NULL,
    created_at DATETIME2 DEFAULT GETDATE()
);
GO

-- ============================================================
-- 3. USERS (with extended image path)
-- ============================================================
CREATE TABLE users (
    user_id INT IDENTITY(1,1) PRIMARY KEY,
    role_id INT NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    status VARCHAR(20) CHECK (status IN ('Active','Inactive','Suspended')) DEFAULT 'Active',
    profile_image_path NVARCHAR(500) NULL,   -- extended length
    created_at DATETIME2 DEFAULT GETDATE(),
    updated_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_users_role FOREIGN KEY (role_id)
        REFERENCES roles(role_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 4. EQUIPMENT (now uses category_id)
-- ============================================================
CREATE TABLE equipment (
    equipment_id INT IDENTITY(1,1) PRIMARY KEY,
    property_number VARCHAR(100) NOT NULL UNIQUE,
    item_name VARCHAR(150) NOT NULL,
    category_id INT NOT NULL,                 -- changed from VARCHAR category
    description VARCHAR(MAX) NULL,            -- was TEXT
    serial_number VARCHAR(100) NULL,
    acquisition_date DATE NULL,
    acquisition_cost DECIMAL(12,2) NULL,
    condition_status VARCHAR(20) CHECK (condition_status IN ('Serviceable','Unserviceable','Under Repair','Disposed')) DEFAULT 'Serviceable',
    created_at DATETIME2 DEFAULT GETDATE(),
    updated_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_equipment_category FOREIGN KEY (category_id)
        REFERENCES categories(category_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 5. REQUESTS
-- ============================================================
CREATE TABLE requests (
    request_id INT IDENTITY(1,1) PRIMARY KEY,
    user_id INT NOT NULL,
    equipment_id INT NOT NULL,
    quantity INT NOT NULL DEFAULT 1,
    purpose VARCHAR(MAX) NOT NULL,            -- was TEXT
    request_status VARCHAR(20) CHECK (request_status IN ('Pending','Approved','Rejected','Cancelled')) DEFAULT 'Pending',
    requested_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_requests_user FOREIGN KEY (user_id)
        REFERENCES users(user_id) ON DELETE NO ACTION,
    CONSTRAINT fk_requests_equipment FOREIGN KEY (equipment_id)
        REFERENCES equipment(equipment_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 6. ASSIGNMENTS
-- ============================================================
CREATE TABLE assignments (
    assignment_id INT IDENTITY(1,1) PRIMARY KEY,
    equipment_id INT NOT NULL,
    user_id INT NOT NULL,
    assigned_date DATE NOT NULL,
    returned_date DATE NULL,
    assignment_status VARCHAR(20) CHECK (assignment_status IN ('Assigned','Returned','Transferred')) DEFAULT 'Assigned',
    remarks VARCHAR(MAX) NULL,                -- was TEXT
    created_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_assignments_equipment FOREIGN KEY (equipment_id)
        REFERENCES equipment(equipment_id) ON DELETE NO ACTION,
    CONSTRAINT fk_assignments_user FOREIGN KEY (user_id)
        REFERENCES users(user_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 7. HANDSHAKES
-- ============================================================
CREATE TABLE handshakes (
    handshake_id INT IDENTITY(1,1) PRIMARY KEY,
    equipment_id INT NOT NULL,
    from_user_id INT NOT NULL,
    to_user_id INT NOT NULL,
    transfer_date DATETIME2 DEFAULT GETDATE(),
    status VARCHAR(20) CHECK (status IN ('Pending','Accepted','Rejected')) DEFAULT 'Pending',
    notes VARCHAR(MAX) NULL,                  -- was TEXT
    CONSTRAINT fk_handshakes_equipment FOREIGN KEY (equipment_id)
        REFERENCES equipment(equipment_id) ON DELETE NO ACTION,
    CONSTRAINT fk_handshakes_from_user FOREIGN KEY (from_user_id)
        REFERENCES users(user_id) ON DELETE NO ACTION,
    CONSTRAINT fk_handshakes_to_user FOREIGN KEY (to_user_id)
        REFERENCES users(user_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 8. DAMAGE REPORTS
-- ============================================================
CREATE TABLE damage_reports (
    report_id INT IDENTITY(1,1) PRIMARY KEY,
    equipment_id INT NOT NULL,
    reported_by INT NOT NULL,
    incident_date DATE NOT NULL,
    damage_description VARCHAR(MAX) NOT NULL, -- was TEXT
    report_status VARCHAR(30) CHECK (report_status IN ('Reported','Under Inspection','Resolved','For Disposal')) DEFAULT 'Reported',
    created_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_damage_equipment FOREIGN KEY (equipment_id)
        REFERENCES equipment(equipment_id) ON DELETE NO ACTION,
    CONSTRAINT fk_damage_reported_by FOREIGN KEY (reported_by)
        REFERENCES users(user_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 9. DISPOSAL REQUESTS
-- ============================================================
CREATE TABLE disposal_requests (
    disposal_id INT IDENTITY(1,1) PRIMARY KEY,
    equipment_id INT NOT NULL,
    requested_by INT NOT NULL,
    reason VARCHAR(MAX) NOT NULL,             -- was TEXT
    disposal_status VARCHAR(30) CHECK (disposal_status IN ('Pending Review','Approved','Completed','Rejected')) DEFAULT 'Pending Review',
    created_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_disposal_equipment FOREIGN KEY (equipment_id)
        REFERENCES equipment(equipment_id) ON DELETE NO ACTION,
    CONSTRAINT fk_disposal_requested_by FOREIGN KEY (requested_by)
        REFERENCES users(user_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 10. NOTIFICATIONS
-- ============================================================
CREATE TABLE notifications (
    notification_id INT IDENTITY(1,1) PRIMARY KEY,
    user_id INT NOT NULL,
    title VARCHAR(150) NOT NULL,
    message VARCHAR(MAX) NOT NULL,            -- was TEXT
    is_read BIT DEFAULT 0,
    created_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_notifications_user FOREIGN KEY (user_id)
        REFERENCES users(user_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 11. ICS DOCUMENTS
-- ============================================================
CREATE TABLE ics_documents (
    ics_id INT IDENTITY(1,1) PRIMARY KEY,
    equipment_id INT NOT NULL,
    issued_to INT NOT NULL,
    ics_number VARCHAR(100) NOT NULL UNIQUE,
    date_issued DATE NOT NULL,
    document_path VARCHAR(500) NULL,          -- increased length
    created_at DATETIME2 DEFAULT GETDATE(),
    CONSTRAINT fk_ics_equipment FOREIGN KEY (equipment_id)
        REFERENCES equipment(equipment_id) ON DELETE NO ACTION,
    CONSTRAINT fk_ics_issued_to FOREIGN KEY (issued_to)
        REFERENCES users(user_id) ON DELETE NO ACTION
);
GO

-- ============================================================
-- 12. INDEXES (for all foreign key columns)
-- ============================================================
CREATE INDEX idx_users_role_id ON users(role_id);
CREATE INDEX idx_equipment_category_id ON equipment(category_id);
CREATE INDEX idx_requests_user_id ON requests(user_id);
CREATE INDEX idx_requests_equipment_id ON requests(equipment_id);
CREATE INDEX idx_assignments_equipment_id ON assignments(equipment_id);
CREATE INDEX idx_assignments_user_id ON assignments(user_id);
CREATE INDEX idx_handshakes_equipment_id ON handshakes(equipment_id);
CREATE INDEX idx_handshakes_from_user_id ON handshakes(from_user_id);
CREATE INDEX idx_handshakes_to_user_id ON handshakes(to_user_id);
CREATE INDEX idx_damage_reports_equipment_id ON damage_reports(equipment_id);
CREATE INDEX idx_damage_reports_reported_by ON damage_reports(reported_by);
CREATE INDEX idx_disposal_requests_equipment_id ON disposal_requests(equipment_id);
CREATE INDEX idx_disposal_requests_requested_by ON disposal_requests(requested_by);
CREATE INDEX idx_notifications_user_id ON notifications(user_id);
CREATE INDEX idx_ics_documents_equipment_id ON ics_documents(equipment_id);
CREATE INDEX idx_ics_documents_issued_to ON ics_documents(issued_to);
GO

-- ============================================================
-- 13. TRIGGERS to auto-update `updated_at` columns
-- ============================================================

-- Users trigger
CREATE TRIGGER trg_users_updated
ON users
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    UPDATE users
    SET updated_at = SYSDATETIME()
    FROM inserted i
    WHERE users.user_id = i.user_id;
END;
GO

-- Equipment trigger
CREATE TRIGGER trg_equipment_updated
ON equipment
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    UPDATE equipment
    SET updated_at = SYSDATETIME()
    FROM inserted i
    WHERE equipment.equipment_id = i.equipment_id;
END;
GO
